#[derive(Clone, Debug, Serialize, Deserialize)]

fn set_clipboard_text(text: &str) {
    unsafe {
        if OpenClipboard(std::ptr::null_mut()) != 0 {
            EmptyClipboard();
            let v: Vec<u16> = text.encode_utf16().chain(std::iter::once(0)).collect();
            let h_mem = GlobalAlloc(GMEM_MOVEABLE, v.len() * 2);
            if !h_mem.is_null() {
                let ptr = GlobalLock(h_mem) as *mut u16;
                std::ptr::copy_nonoverlapping(v.as_ptr(), ptr, v.len());
                GlobalUnlock(h_mem);
                if SetClipboardData(CF_UNICODETEXT, h_mem).is_null() {
                    GlobalFree(h_mem);
                }
            }
            CloseClipboard();
        }
    }
}

fn get_clipboard_text() -> Option<String> {
    unsafe {
        if OpenClipboard(std::ptr::null_mut()) != 0 {
            let h_mem = GetClipboardData(CF_UNICODETEXT);
            if !h_mem.is_null() {
                let ptr = GlobalLock(h_mem) as *const u16;
                if !ptr.is_null() {
                    let mut len = 0;
                    while *ptr.add(len) != 0 {
                        len += 1;
                    }
                    let slice = std::slice::from_raw_parts(ptr, len);
                    let res = String::from_utf16(slice).ok();
                    GlobalUnlock(h_mem);
                    CloseClipboard();
                    return res;
                }
                GlobalUnlock(h_mem);
            }
            CloseClipboard();
        }
    }
    None
}

fn build_tree(
    pid: i32,
    depth: i32,
    db: &Vec<RawItem>,
    v_items: &mut Vec<MyTreeItem>,
    hidden: bool,
    parent_lasts: Vec<bool>,
    sel_id: i32,
    _parent_coord: Point2D,
    global_index: &mut i32,
) {
    let mut children: Vec<&RawItem> = db.iter().filter(|i| i.parent_id == pid).collect();
    children.sort_by_key(|i| i.id);
    let total_children = children.len();

    for (index, item) in children.into_iter().enumerate() {
        let is_last = index == total_children - 1;
        let has_children = db.iter().any(|i| i.parent_id == item.id);
        let row_height = 32.0_f32;
        let padding_top = 8.0_f32;
        let padding_left = 8.0_f32;

        let current_row_idx = *global_index;
        if !hidden {
            *global_index += 1;
        }

        let self_x = padding_left + 10.0 + (depth as f32 * 24.0) + 10.0;
        let self_y = padding_top + (current_row_idx as f32 * row_height) + 16.0;
        let current_node_coord = Point2D {
            x: self_x,
            y: self_y,
        };

        if !hidden {
            let text_len = item.text.chars().count() as f32;
            let computed_w = (text_len * 7.2) + 20.0;
            let last_flag = if is_last { "L" } else { "M" };
            let smart_id = format!(
                "{}_{}_{}_{}_{}",
                item.parent_id,
                item.id,
                depth,
                last_flag,
                item.links.len()
            );

            let mut slint_mask = [0; 12];
            for i in 0..12 {
                if i < (depth - 1) as usize && i < parent_lasts.len() {
                    slint_mask[i] = if !parent_lasts[i] { 1 } else { 0 };
                }
            }

            v_items.push(MyTreeItem {
                id: item.id,
                text: item.text.clone().into(),
                depth,
                is_last,
                has_children,
                is_editing: item.is_editing,
                is_expanded: item.is_expanded,
                parent_is_last: depth > 1 && parent_lasts[(depth - 2) as usize],
                is_selected: item.id == sel_id,
                note: item.note.clone().into(),
                text_width: computed_w,
                smart_id: smart_id.into(),
                line_mask: ModelRc::from(Rc::new(VecModel::from(slint_mask.to_vec()))),
                parent_id: item.parent_id,
                parent_triangle_x: _parent_coord.x,
                parent_triangle_y: _parent_coord.y,
                self_triangle_x: current_node_coord.x,
                self_triangle_y: current_node_coord.y,
            });
        }

        if has_children {
            let mut current_lasts = parent_lasts.clone();
            current_lasts.push(is_last);
            build_tree(
                item.id,
                depth + 1,
                db,
                v_items,
                hidden || !item.is_expanded,
                current_lasts,
                sel_id,
                current_node_coord,
                global_index,
            );
        }
    }
}
fn update_ui_models(app: &AppWindow, state: &AppState) {
    let mut t1_views = Vec::new();
    let mut g_idx1 = 0;
    build_tree(
        0,
        0,
        &state.tree1_items,
        &mut t1_views,
        false,
        vec![],
        state.selected_tree1_id,
        Point2D { x: 20.0, y: 24.0 },
        &mut g_idx1,
    );

    let mut t2_views = Vec::new();
    let mut g_idx2 = 0;
    build_tree(
        0,
        0,
        &state.tree2_items,
        &mut t2_views,
        false,
        vec![],
        state.selected_tree2_id,
        Point2D { x: 20.0, y: 24.0 },
        &mut g_idx2,
    );

    let mut max_width1 = 250.0_f32;
    for item in &t1_views {
        let item_width = 10.0 + (item.depth as f32 * 24.0) + 20.0 + item.text_width + 40.0;
        max_width1 = max_width1.max(item_width);
    }
    app.set_tree1_width(max_width1);

    let mut max_width2 = 250.0_f32;
    for item in &t2_views {
        let item_width = 10.0 + (item.depth as f32 * 24.0) + 20.0 + item.text_width + 40.0;
        max_width2 = max_width2.max(item_width);
    }
    app.set_tree2_width(max_width2);

    // Новое: передаем количество видимых элементов для расчета динамической высоты в Slint
    app.set_tree1_count(g_idx1);
    app.set_tree2_count(g_idx2);

    app.set_tree1_data(ModelRc::from(Rc::new(VecModel::from(t1_views.clone()))));
    app.set_tree2_data(ModelRc::from(Rc::new(VecModel::from(t2_views.clone()))));

    let t1_cad_x = app.get_tree1_cad_x();
    let t1_cad_y = app.get_tree1_cad_y();
    let t2_cad_x = app.get_tree2_cad_x();
    let t2_cad_y = app.get_tree2_cad_y();

    let mut updated_links = Vec::new();
    for link in &state.cad_links {
        let mut line = link.clone();
        let mut found_src = false;
        let mut found_dst = false;

        if let Some(item) = t1_views.iter().find(|i| i.id == link.id_obj1) {
            line.x1 = t1_cad_x + item.self_triangle_x + 20.0 + item.text_width;
            line.y1 = t1_cad_y + item.self_triangle_y;
            found_src = true;
        } else if let Some(item) = t2_views.iter().find(|i| i.id == link.id_obj1) {
            line.x1 = t2_cad_x + item.self_triangle_x;
            line.y1 = t2_cad_y + item.self_triangle_y;
            found_src = true;
        }

        if let Some(item) = t1_views.iter().find(|i| i.id == link.id_obj2) {
            line.x2 = t1_cad_x + item.self_triangle_x + 20.0 + item.text_width;
            line.y2 = t1_cad_y + item.self_triangle_y;
            found_dst = true;
        } else if let Some(item) = t2_views.iter().find(|i| i.id == link.id_obj2) {
            line.x2 = t2_cad_x + item.self_triangle_x;
            line.y2 = t2_cad_y + item.self_triangle_y;
            found_dst = true;
        }
        if found_src && found_dst {
            updated_links.push(line);
        }
    }
    app.set_active_cad_links(ModelRc::from(Rc::new(VecModel::from(updated_links))));
}

fn pack_branch(pid: i32, vec: &Vec<RawItem>) -> Vec<ClipboardBranch> {
    vec.iter()
        .filter(|i| i.parent_id == pid)
        .map(|i| ClipboardBranch {
            root_item: i.clone(),
            children: pack_branch(i.id, vec),
        })
        .collect()
}

fn unpack_branch(branch: ClipboardBranch, new_parent_id: i32, vec: &mut Vec<RawItem>) {
    let next_id = vec.iter().map(|i| i.id).max().unwrap_or(0) + 1;
    let mut new_item = branch.root_item;
    new_item.id = next_id;
    new_item.parent_id = new_parent_id;
    new_item.is_editing = false;
    vec.push(new_item);
    for child in branch.children {
        unpack_branch(child, next_id, vec);
    }
}

        } else {
            &mut st.tree2_items
        };
        if let Some(item) = target.iter_mut().find(|i| i.id == id) {
            item.is_expanded = !item.is_expanded;
        }
        update_ui_models(&a_weak.unwrap(), &st);
    });

    let _s_clone = state.clone();
    let a_weak = app_weak.clone();
    app.on_trees_geometry_changed(move || {
        update_ui_models(&a_weak.unwrap(), &_s_clone.borrow());
    });

    let _s_clone = state.clone();
    let a_weak = app_weak.clone();
    app.on_add_cad_link_mode(move |id, _t_idx| {
        let ui = a_weak.unwrap();
        ui.set_is_linking_mode(true);
        ui.set_linking_source_node_id(id);
    });

    let s_clone = state.clone();
    let a_weak = app_weak.clone();
    app.on_hover_item(move |id, index, tree_idx| {
        let ui = a_weak.unwrap();
        let st = s_clone.borrow();
        if id != -1 {
            let target = if tree_idx == 1 {
                &st.tree1_items
            } else {
                &st.tree2_items
            };
            if let Some(item) = target.iter().find(|i| i.id == id) {
                let display_note = if item.note.is_empty() {
                    "Примечание отсутствует.".to_string()
                } else {
                    item.note.clone()
                };
                ui.set_tooltip_text(display_note.into());
                ui.set_tooltip_index(index);
            }
        } else {
            ui.set_tooltip_index(-1);
        }
    });

    let s_clone = state.clone();
    let a_weak = app_weak.clone();
    app.on_select_item(move |id, _text, tree_idx| {
        let ui = a_weak.unwrap();
        let mut st = s_clone.borrow_mut();
        st.active_focus_tree = tree_idx;
        if tree_idx == 1 {
            st.selected_tree1_id = id;
        } else {
            st.selected_tree2_id = id;
        }

        if ui.get_is_linking_mode() {
            let src_id = ui.get_linking_source_node_id();
            if src_id != id {
                let link_id = st
                    .tree1_items
                    .iter()
                    .chain(st.tree2_items.iter())
                    .map(|i| i.id)
                    .max()
                    .unwrap_or(0)
                    + 1;
                st.cad_links.push(CADLine {
                    id: link_id,
                    id_obj1: src_id,
                    id_obj2: id,
                    x1: 0.0,
                    y1: 0.0,
                    x2: 0.0,
                    y2: 0.0,
                });
                ui.set_is_linking_mode(false);
            }
        }
        if let Some(item) = (if tree_idx == 1 {
            &st.tree1_items
        } else {
            &st.tree2_items
        })
        .iter()
        .find(|i| i.id == id)
        {
            ui.set_properties_text(item.note.clone().into());
        }
        update_ui_models(&ui, &st);

    let s_clone = state.clone();
    let a_weak = app_weak.clone();
    app.on_key_esc_pressed(move || {
        let ui = a_weak.unwrap();
        ui.set_is_linking_mode(false);
        update_ui_models(&ui, &s_clone.borrow());
    });

    let s_clone = state.clone();
    app.on_change_note(move |id, new_note, tree_idx| {
        let mut st = s_clone.borrow_mut();
        let target = if tree_idx == 1 {
            &mut st.tree1_items
        } else {
            &mut st.tree2_items
        };
        if let Some(item) = target.iter_mut().find(|i| i.id == id) {
            item.note = new_note.to_string();
        }
    });

    let s_clone = state.clone();
    let a_weak = app_weak.clone();

        let ui = a_weak.unwrap();
        let mut st = s_clone.borrow_mut();
        let target = if tree_idx == 1 {
            &mut st.tree1_items
        } else {
            &mut st.tree2_items
        };
        for item in target.iter_mut() {
            item.is_editing = item.id == id;

        update_ui_models(&ui, &st);

    let a_weak = app_weak.clone();
    app.on_finish_rename(move |id, new_text, tree_idx| {
        let ui = a_weak.unwrap();
        let mut st = s_clone.borrow_mut();
        let target = if tree_idx == 1 {
            &mut st.tree1_items
        } else {
            &mut st.tree2_items

        if let Some(item) = target.iter_mut().find(|i| i.id == id) {
            item.text = new_text.to_string();
            item.is_editing = false;
        }
        update_ui_models(&ui, &st);
    });

    let s_clone = state.clone();
    let a_weak = app_weak.clone();
    app.on_menu_action(move |action, target_id, tree_idx| {
        let ui = a_weak.unwrap();
        let (old_items, new_id) = {
            let st = s_clone.borrow();
            let old = if tree_idx == 1 {
                st.tree1_items.clone()
            } else {
                st.tree2_items.clone()
            };
            let max_id = st
                .tree1_items
                .iter()
                .chain(st.tree2_items.iter())
                .map(|i| i.id)
                .max()
                .unwrap_or(0);
            (old, max_id + 1)
        };

        let mut st = s_clone.borrow_mut();
        match action.as_str() {
            "add" => {
                if tree_idx == 1 {
                    st.history_tree1.push(old_items);
                } else {
                    st.history_tree2.push(old_items);
                }
                let target = if tree_idx == 1 {
                    &mut st.tree1_items
                } else {
                    &mut st.tree2_items
                };
                if let Some(p) = target.iter_mut().find(|i| i.id == target_id) {
                    p.is_expanded = true;
                }
                target.push(RawItem {
                    id: new_id,
                    text: format!("Новый узел {}", new_id),
                    parent_id: target_id,
                    is_expanded: true,
                    is_editing: true,
                    note: "".into(),
                    links: vec![],
                });
                if tree_idx == 1 {
                    st.selected_tree1_id = new_id;
                } else {
                    st.selected_tree2_id = new_id;
                }
            }
            "delete" => {
                if tree_idx == 1 {
                    st.history_tree1.push(old_items);
                } else {
                    st.history_tree2.push(old_items);
                }
                fn del_rec(pid: i32, vec: &mut Vec<RawItem>) {
                    let sub: Vec<i32> = vec
                        .iter()
                        .filter(|i| i.parent_id == pid)
                        .map(|i| i.id)
                        .collect();
                    for s in sub {
                        del_rec(s, vec);
                    }
                    vec.retain(|i| i.id != pid);
                }
                let target = if tree_idx == 1 {
                    &mut st.tree1_items
                } else {
                    &mut st.tree2_items
                };
                if target_id != 1 && target_id != 10 {
                    del_rec(target_id, target);
                }
                st.cad_links
                    .retain(|l| l.id_obj1 != target_id && l.id_obj2 != target_id);
            }
            "rename" => {
                let target = if tree_idx == 1 {
                    &mut st.tree1_items
                } else {
                    &mut st.tree2_items
                };
                if let Some(item) = target.iter_mut().find(|i| i.id == target_id) {
                    item.is_editing = true;
                }
            }
            "properties" => {
                let target = if tree_idx == 1 {
                    &st.tree1_items
                } else {
                    &st.tree2_items
                };
                if let Some(item) = target.iter().find(|i| i.id == target_id) {
                    ui.set_properties_text(item.note.clone().into());
                    ui.set_properties_visible(true);
                }
            }
            _ => {}
        }
        update_ui_models(&ui, &st);
    });

    let s_clone = state.clone();
    let a_weak = app_weak.clone();
    app.on_undo_action(move || {
        let mut st = s_clone.borrow_mut();
        if st.active_focus_tree == 1 {
            if let Some(prev) = st.history_tree1.pop() {
                st.tree1_items = prev;
            }
        } else {
            if let Some(prev) = st.history_tree2.pop() {
                st.tree2_items = prev;
            }
        }
        update_ui_models(&a_weak.unwrap(), &st);
    });

    let s_clone = state.clone();
    app.on_copy_action(move || {
        let st = s_clone.borrow();
        let tree_idx = st.active_focus_tree;
        let target = if tree_idx == 1 {
            &st.tree1_items
        } else {
            &st.tree2_items
        };
        let sel = if tree_idx == 1 {
            st.selected_tree1_id
        } else {
            st.selected_tree2_id
        };
        if let Some(item) = target.iter().find(|i| i.id == sel) {
            if item.is_editing {
                set_clipboard_text(&item.text);
            } else {
                let branch = ClipboardBranch {
                    root_item: item.clone(),
                    children: pack_branch(item.id, target),
                };
                if let Ok(json) = serde_json::to_string(&branch) {
                    set_clipboard_text(&format!("TREE_BRANCH_DATA:{}", json));
                }
            }
        }
    });

    let s_clone = state.clone();
    let a_weak = app_weak.clone();
    app.on_paste_action(move || {
        if let Some(txt) = get_clipboard_text() {
            let (t_idx, sel, is_edit, old_items) = {
                let st = s_clone.borrow();
                let t_idx = st.active_focus_tree;
                let sel = if t_idx == 1 {
                    st.selected_tree1_id
                } else {
                    st.selected_tree2_id
                };
                let target = if t_idx == 1 {
                    &st.tree1_items
                } else {
                    &st.tree2_items
                };
                let is_edit = target
                    .iter()
                    .find(|i| i.id == sel)
                    .map(|i| i.is_editing)
                    .unwrap_or(false);
                (t_idx, sel, is_edit, target.clone())
            };

            let mut st = s_clone.borrow_mut();
            if is_edit {
                let target = if t_idx == 1 {
                    &mut st.tree1_items
                } else {
                    &mut st.tree2_items
                };
                if let Some(i) = target.iter_mut().find(|i| i.id == sel) {
                    i.text = txt;
                }
            } else if txt.starts_with("TREE_BRANCH_DATA:") {
                if t_idx == 1 {
                    st.history_tree1.push(old_items);
                } else {
                    st.history_tree2.push(old_items);
                }
                let json = txt.chars().skip(17).collect::<String>();
                if let Ok(br) = serde_json::from_str::<ClipboardBranch>(&json) {
                    let target = if t_idx == 1 {
                        &mut st.tree1_items
                    } else {
                        &mut st.tree2_items
                    };
                    unpack_branch(br, sel, target);
                    if let Some(p) = target.iter_mut().find(|i| i.id == sel) {
                        p.is_expanded = true;
                    }
                }
            }
            update_ui_models(&a_weak.unwrap(), &st);
        }
    });

    let app_for_win = app.as_weak();
    slint::invoke_from_event_loop(move || {
        if let Some(_ui) = app_for_win.upgrade() {
            let title: Vec<u16> = "Менеджер команд голосового управления (CAD Среда)\0"
                .encode_utf16()
                .collect();
            unsafe {
                let hwnd = FindWindowW(std::ptr::null(), title.as_ptr());
                if !hwnd.is_null() {
                    let cur = GetWindowLongW(hwnd, GWL_STYLE);
                    SetWindowLongW(
                        hwnd,
                        GWL_STYLE,
                        cur | WS_CAPTION | WS_MINIMIZEBOX | WS_MAXIMIZEBOX | WS_THICKFRAME,
                    );
                    SetWindowPos(
                        hwnd,
                        std::ptr::null_mut(),
                        0,
                        0,
                        0,
                        0,
                        SWP_FRAMECHANGED | SWP_NOMOVE | SWP_NOSIZE | SWP_NOZORDER,
                    );
                    ShowWindow(hwnd, 3);
                }
            }
        }
    })
    .unwrap();

    spawn_hotkey_thread(app.as_weak());
    app.run().unwrap();
}

fn spawn_hotkey_thread(app_weak: Weak<AppWindow>) {
    thread::spawn(move || {
        let mut ctrl_was = false;
        let mut z_was = false;
        let mut c_was = false;
        let mut v_was = false;
        let title: Vec<u16> = "Менеджер команд голосового управления (CAD Среда)\0"
            .encode_utf16()
            .collect();
        let mut hw: *mut std::ffi::c_void = std::ptr::null_mut();
        loop {
            thread::sleep(Duration::from_millis(50));
            unsafe {
                if hw.is_null() {
                    hw = FindWindowW(std::ptr::null(), title.as_ptr());
                }
                let active = GetForegroundWindow();
                if !hw.is_null() && (active == hw) {
                    let ctrl = (GetAsyncKeyState(0x11) & -32768_i16) != 0;
                    let z = (GetAsyncKeyState(0x5A) & -32768_i16) != 0;
                    let c = (GetAsyncKeyState(0x43) & -32768_i16) != 0;
                    let v = (GetAsyncKeyState(0x56) & -32768_i16) != 0;
                    if ctrl && z && (!ctrl_was || !z_was) {
                        let aw = app_weak.clone();
                        let _ = slint::invoke_from_event_loop(move || {
                            if let Some(a) = aw.upgrade() {
                                a.invoke_undo_action();
                            }
                        });
                    }
                    if ctrl && c && (!ctrl_was || !c_was) {
                        let aw = app_weak.clone();
                        let _ = slint::invoke_from_event_loop(move || {
                            if let Some(a) = aw.upgrade() {
                                a.invoke_copy_action();
                            }
                        });
                    }
                    if ctrl && v && (!ctrl_was || !v_was) {
                        let aw = app_weak.clone();
                        let _ = slint::invoke_from_event_loop(move || {
                            if let Some(a) = aw.upgrade() {
                                a.invoke_paste_action();
                            }
                        });
                    }
                    ctrl_was = ctrl;
                    z_was = z;
                    c_was = c;
                    v_was = v;
                }
            }
        }
    });
}
