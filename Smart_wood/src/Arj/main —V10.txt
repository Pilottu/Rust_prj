#[link(name = "kernel32")]
extern "system" {
 fn GlobalAlloc(u_flags: u32, dw_bytes: usize) -> *mut std::ffi::c_void;
 fn GlobalLock(h_mem: *mut std::ffi::c_void) -> *mut std::ffi::c_void;

}

const GWL_STYLE: i32 = -16;
const WS_CAPTION: i32 = 0x00C00000;
const WS_MAXIMIZEBOX: i32 = 0x00010000;
const WS_MINIMIZEBOX: i32 = 0x00020000;
const WS_THICKFRAME: i32 = 0x00040000;
const SWP_FRAMECHANGED: u32 = 0x0020;
const SWP_NOMOVE: u32 = 0x0002;
const SWP_NOSIZE: u32 = 0x0001;
const SWP_NOZORDER: u32 = 0x0004;
const CF_UNICODETEXT: u32 = 13;
const GMEM_MOVEABLE: u32 = 0x0002;

#[allow(dead_code)]
#[derive(Clone, Debug)]
struct RawLink {
 id: i32,
 target_node_id: i32,
 target_node_name: String,
}

#[allow(dead_code)]
#[derive(Clone, Debug)]
struct RawItem {
 id: i32,
 text: String,
 parent_id: i32,
 is_expanded: bool,
 is_editing: bool,
 note: String,
 links: Vec<RawLink>,
}

#[derive(Clone, Copy)]
struct Point2D {
 x: f32,
 y: f32,
}

fn set_clipboard_text(text: &str) {
 unsafe {
 if OpenClipboard(std::ptr::null_mut()) != 0 {
 EmptyClipboard();
 let v: Vec<u16> = text.encode_utf16().chain(std::iter::once(0)).collect();
 let h_mem = GlobalAlloc(GMEM_MOVEABLE, v.len() * 2);
 if !h_mem.is_null() {
 let ptr = GlobalLock(h_mem) as *mut u16;
 std::ptr::copy_nonoverlapping(v.as_ptr(), ptr, v.len());
 GlobalUnlock(h_mem);
 if SetClipboardData(CF_UNICODETEXT, h_mem).is_null() {
 GlobalFree(h_mem);
 }
 }
 CloseClipboard();
 }
 }
}

fn get_clipboard_text() -> Option<String> {
 unsafe {
 if OpenClipboard(std::ptr::null_mut()) != 0 {
 let h_mem = GetClipboardData(CF_UNICODETEXT);
 if !h_mem.is_null() {
 let ptr = GlobalLock(h_mem) as *const u16;
 if !ptr.is_null() {
 let mut len = 0;
 while *ptr.add(len) != 0 {
 len += 1;
 }
 let slice = std::slice::from_raw_parts(ptr, len);
 let res = String::from_utf16(slice).ok();
 GlobalUnlock(h_mem);
 CloseClipboard();
 return res;
 }
 GlobalUnlock(h_mem);
 }
 CloseClipboard();
 }
 }
 None
}
fn spawn_hotkey_thread(app_weak: Weak<AppWindow>) {
 thread::spawn(move || {
 let mut ctrl_was_down = false;
 let mut z_was_down = false;
 let mut c_was_down = false;
 let mut v_was_down = false;
 let title: Vec<u16> = "Менеджер команд голосового управления (Режим Графа Связей)\0".encode_utf16().collect();
 let mut my_hwnd: *mut std::ffi::c_void = std::ptr::null_mut();
 loop {
 thread::sleep(Duration::from_millis(50));
 unsafe {
 if my_hwnd.is_null() {
 my_hwnd = FindWindowW(std::ptr::null(), title.as_ptr());
 }
 let active_hwnd = GetForegroundWindow();
 let is_window_active = !my_hwnd.is_null() && (active_hwnd == my_hwnd);
 let ctrl_down = (GetAsyncKeyState(0x11) & -32768_i16) != 0;
 let z_down = (GetAsyncKeyState(0x5A) & -32768_i16) != 0;
 let c_down = (GetAsyncKeyState(0x43) & -32768_i16) != 0;
 let v_down = (GetAsyncKeyState(0x56) & -32768_i16) != 0;
 if is_window_active && ctrl_down && z_down && (!ctrl_was_down || !z_was_down) {
 let aw = app_weak.clone();
 let _ = slint::invoke_from_event_loop(move || { if let Some(app) = aw.upgrade() { app.invoke_undo_action(); } });
 }
 if is_window_active && ctrl_down && c_down && (!ctrl_was_down || !c_was_down) {
 let aw = app_weak.clone();
 let _ = slint::invoke_from_event_loop(move || { if let Some(app) = aw.upgrade() { app.invoke_copy_action(); } });
 }
 if is_window_active && ctrl_down && v_down && (!ctrl_was_down || !v_was_down) {
 let aw = app_weak.clone();
 let _ = slint::invoke_from_event_loop(move || { if let Some(app) = aw.upgrade() { app.invoke_paste_action(); } });
 }
 ctrl_was_down = ctrl_down; z_was_down = z_down; c_was_down = c_down; v_was_down = v_down;
 }
 }
 });
}

fn build_tree(
 pid: i32,
 depth: i32,
 db: &Vec<RawItem>,
 v_items: &mut Vec<MyTreeItem>,
 hidden: bool,
 parent_lasts: Vec<bool>,
 sel_id: i32,
 parent_coord: Point2D,
 global_index: &mut i32,
) {
 let mut children: Vec<&RawItem> = db.iter().filter(|i| i.parent_id == pid).collect();
 children.sort_by_key(|i| i.id);
 let total_children = children.len();
 for (index, item) in children.into_iter().enumerate() {
 let is_last = index == total_children - 1;
 let has_children = db.iter().any(|i| i.parent_id == item.id);
 let row_height = 32.0_f32;
 let padding_top = 8.0_f32;
 let padding_left = 8.0_f32;
 let current_row_idx = *global_index;
 if !hidden { *global_index += 1; }
 let self_x = padding_left + 10.0 + (depth as f32 * 24.0) + 10.0;
 let self_y = padding_top + (current_row_idx as f32 * row_height) + 16.0;
 let current_node_coord = Point2D { x: self_x, y: self_y };
 if !hidden {
 let text_len = item.text.chars().count() as f32;
 let computed_w = (text_len * 7.2) + 20.0;
 let last_flag = if is_last { "L" } else { "M" };
 let smart_id = format!("{}_{}_{}_{}_{}", item.parent_id, item.id, depth, last_flag, item.links.len());
 let mut slint_mask = [0; 6];
 for i in 0..6 { if i < (depth - 1) as usize && i < parent_lasts.len() { slint_mask[i] = if !parent_lasts[i] { 1 } else { 0 }; } }
 v_items.push(MyTreeItem {
 id: item.id, text: item.text.clone().into(), depth, is_last, has_children, is_editing: item.is_editing, is_expanded: item.is_expanded, parent_is_last: depth > 1 && parent_lasts[(depth - 2) as usize], is_selected: item.id == sel_id, note: item.note.clone().into(), text_width: computed_w, smart_id: smart_id.into(), line_mask: Rc::new(VecModel::from(slint_mask.to_vec())).into(), parent_id: item.parent_id, parent_triangle_x: parent_coord.x, parent_triangle_y: parent_coord.y, self_triangle_x: current_node_coord.x, self_triangle_y: current_node_coord.y,
 });
 }
 if has_children {
 let mut current_lasts = parent_lasts.clone(); current_lasts.push(is_last);
 build_tree(item.id, depth + 1, db, v_items, hidden || !item.is_expanded, current_lasts, sel_id, current_node_coord, global_index);
 }
 }
}
fn main() -> Result<(), slint::PlatformError> {
 let app = AppWindow::new()?;
 app.set_scale_factor(1.0_f32);
 slint::invoke_from_event_loop(move || {
 let title: Vec<u16> = "Менеджер команд голосового управления (Режим Графа Связей)\0".encode_utf16().collect();
 unsafe {
 let hwnd = FindWindowW(std::ptr::null(), title.as_ptr());
 if !hwnd.is_null() {
 let cur = GetWindowLongW(hwnd, GWL_STYLE);
 SetWindowLongW(hwnd, GWL_STYLE, cur | WS_CAPTION | WS_MINIMIZEBOX | WS_MAXIMIZEBOX | WS_THICKFRAME);
 SetWindowPos(hwnd, std::ptr::null_mut(), 0, 0, 0, 0, SWP_FRAMECHANGED | SWP_NOMOVE | SWP_NOSIZE | SWP_NOZORDER);
 ShowWindow(hwnd, 3);
 }
 }
 }).unwrap();
 spawn_hotkey_thread(app.as_weak());

 let selected_id = Rc::new(RefCell::new(1_i32));
 let selected_right_id = Rc::new(RefCell::new(-1_i32));
 let all_items = Rc::new(RefCell::new(vec![RawItem { id: 1, text: "Основной узел".into(), parent_id: 0, is_expanded: true, is_editing: false, note: "Главный сервер управления".into(), links: vec![] }]));
 let history: Rc<RefCell<Vec<Vec<RawItem>>>> = Rc::new(RefCell::new(Vec::new()));

 let update_view = {
 let app_weak = app.as_weak(); let all_items = all_items.clone(); let selected_id = selected_id.clone(); let selected_right_id = selected_right_id.clone();
 move || {
 if let Some(app) = app_weak.upgrade() {
 let db = all_items.borrow();
 let sel_id = *selected_id.borrow();
 let sel_r_id = *selected_right_id.borrow();
 
 let mut left_items = Vec::new(); let mut counter_l = 0;
 build_tree(0, 0, &db, &mut left_items, false, vec![], sel_id, Point2D { x: 20.0, y: 24.0 }, &mut counter_l);
 
 let mut right_items = Vec::new(); let mut counter_r = 0;
 build_tree(0, 0, &db, &mut right_items, false, vec![], sel_r_id, Point2D { x: 20.0, y: 24.0 }, &mut counter_r);

 let mut max_w = 250.0_f32;
 for item in &left_items { max_w = max_w.max(10.0 + (item.depth as f32 * 24.0) + 20.0 + item.text_width + 40.0); }
 
 app.set_tree_width(max_w.into());
 app.set_tree_width_right(max_w.into());
 app.set_tree_data(ModelRc::from(Rc::new(VecModel::from(left_items.clone()))));
 app.set_tree_data_right(ModelRc::from(Rc::new(VecModel::from(right_items.clone()))));

          // --- ДИНАМИЧЕСКИЙ РАСЧЕТ МЕЖПАНЕЛЬНОГО Z-ЩУПАЛЬЦА В ЧИСТЫХ CAD КООРДИНАТАХ ---
         if app.get_link_mode_active() && sel_id != -1 && sel_r_id != -1 {
             if let (Some(l_item), Some(r_item)) = (left_items.iter().find(|i| i.id == sel_id), right_items.iter().find(|i| i.id == sel_r_id)) {
                 
                 // Точка старта: правый край овала левого дерева (учитываем паддинг ListView 8px)
                 let start_x = 8.0 + l_item.self_triangle_x + 12.0 + l_item.text_width + 20.0;
                 let start_y = 8.0 + l_item.self_triangle_y + 4.0;
                 
                 // Расстояние между панелями в виртуальном пространстве (ширина дерева + 80px межпанельного интервала)
                 let panel_spacing = max_w + 80.0;
                 
                 // Точка финиша: левый край овала правого дерева
                 let end_x = panel_spacing + 8.0 + r_item.self_triangle_x + 12.0;
                 let end_y = 8.0 + r_item.self_triangle_y + 4.0;

                 // Передаем в Slint чистые декартовы координаты поля
                 app.set_link_start_x(start_x); app.set_link_start_y(start_y);
                 app.set_link_end_x(end_x); app.set_link_end_y(end_y);
             }
         } else {
             app.set_link_start_x(0.0); app.set_link_end_x(0.0);
         }

 }
 }
 };

 let update_view_clone = update_view.clone(); let all_items_toggle = all_items.clone();
 app.on_toggle_expand(move |id| { if let Some(item) = all_items_toggle.borrow_mut().iter_mut().find(|i| i.id == id) { item.is_expanded = !item.is_expanded; } update_view_clone(); });

 let update_view_clone_r = update_view.clone(); let all_items_toggle_r = all_items.clone();
 app.on_toggle_expand_right(move |id| { if let Some(item) = all_items_toggle_r.borrow_mut().iter_mut().find(|i| i.id == id) { item.is_expanded = !item.is_expanded; } update_view_clone_r(); });

 let selected_id_clone = selected_id.clone(); let update_view_clone3 = update_view.clone();
 app.on_select_item(move |id, _txt| { *selected_id_clone.borrow_mut() = id; update_view_clone3(); });

 let selected_right_id_clone = selected_right_id.clone(); let update_view_clone3_r = update_view.clone();
 app.on_select_right_item(move |id, _txt| { *selected_right_id_clone.borrow_mut() = id; update_view_clone3_r(); });

 let app_weak_cancel = app.as_weak(); let update_view_cancel = update_view.clone(); let sel_r_cancel = selected_right_id.clone();
 app.on_cancel_link_mode(move || { if let Some(app) = app_weak_cancel.upgrade() { app.set_link_mode_active(false); *sel_r_cancel.borrow_mut() = -1; update_view_cancel(); } });

 let app_weak_menu = app.as_weak();
 app.on_show_menu(move |id, x, y| { if let Some(app) = app_weak_menu.upgrade() { if !app.get_link_mode_active() { app.set_menu_target_id(id); app.set_menu_x(x); app.set_menu_y(y); app.set_menu_visible(true); } } });

 let app_weak_hover = app.as_weak(); let all_items_hover = all_items.clone();
 app.on_hover_item(move |id, index, _x, _y| { if let Some(a) = app_weak_hover.upgrade() { if id != -1 { if let Some(item) = all_items_hover.borrow().iter().find(|i| i.id == id) { a.set_tooltip_text(if item.note.is_empty() { "Примечание отсутствует.".into() } else { item.note.clone().into() }); a.set_tooltip_index(index); } } else { a.set_tooltip_index(-1); } } });

 let app_weak_action = app.as_weak(); let all_items_action = all_items.clone(); let history_action = history.clone(); let update_view_action = update_view.clone(); let selected_id_action = selected_id.clone();
 app.on_menu_action(move |action_type, target_id| {
 if let Some(app) = app_weak_action.upgrade() {
 app.set_menu_visible(false);
 if action_type == "link_mode" {
 app.set_link_mode_active(true);
 app.set_link_source_node_id(target_id);
 update_view_action();
 } else if action_type == "properties" {
 if let Some(item) = all_items_action.borrow().iter().find(|i| i.id == target_id) { app.set_properties_text(item.note.clone().into()); app.set_properties_visible(true); }
 } else if action_type == "add" {
 history_action.borrow_mut().push(all_items_action.borrow().clone());
 let mut db = all_items_action.borrow_mut(); if let Some(parent) = db.iter_mut().find(|i| i.id == target_id) { parent.is_expanded = true; }
 let new_id = db.iter().map(|i| i.id).max().unwrap_or(0) + 1;
 db.push(RawItem { id: new_id, text: format!("Новый узел {}", new_id), parent_id: target_id, is_expanded: true, is_editing: true, note: "".into(), links: vec![] });
 *selected_id_action.borrow_mut() = new_id; drop(db); update_view_action();
 } else if action_type == "rename" {
 if let Some(item) = all_items_action.borrow_mut().iter_mut().find(|i| i.id == target_id) { item.is_editing = true; } update_view_action();
 } else if action_type == "delete" {
 history_action.borrow_mut().push(all_items_action.borrow().clone());
 let mut db = all_items_action.borrow_mut();
 fn delete_recursive(pid: i32, vec: &mut Vec<RawItem>) {
 let sub_ids = vec.iter().filter(|i| i.parent_id == pid).map(|i| i.id).collect::<Vec<i32>>();
 for sid in sub_ids { delete_recursive(sid, vec); }
 vec.retain(|i| i.id != pid);
 }
 if target_id != 1 { delete_recursive(target_id, &mut db); } drop(db); update_view_action();
 }
 }
 });

 let all_items_rename = all_items.clone(); let update_view_rename = update_view.clone();
 app.on_finish_rename(move |id, new_text| { if let Some(item) = all_items_rename.borrow_mut().iter_mut().find(|i| i.id == id) { item.text = new_text.to_string(); item.is_editing = false; } update_view_rename(); });

 let all_items_copy = all_items.clone(); let selected_id_copy = selected_id.clone();
 app.on_copy_action(move || { if let Some(item) = all_items_copy.borrow().iter().find(|i| i.id == *selected_id_copy.borrow()) { set_clipboard_text(&item.text); } });

 let all_items_paste = all_items.clone(); let selected_id_paste = selected_id.clone(); let history_paste = history.clone(); let update_view_paste = update_view.clone();
 app.on_paste_action(move || { if let Some(clip_text) = get_clipboard_text() { history_paste.borrow_mut().push(all_items_paste.borrow().clone()); if let Some(item) = all_items_paste.borrow_mut().iter_mut().find(|i| i.id == *selected_id_paste.borrow()) { if !item.is_editing { item.text = clip_text; } } update_view_paste(); } });

 let all_items_note = all_items.clone(); app.on_change_note(move |id, new_note| { if let Some(item) = all_items_note.borrow_mut().iter_mut().find(|i| i.id == id) { item.note = new_note.to_string(); } });

 let all_items_dbl = all_items.clone(); let update_view_dbl = update_view.clone();
 app.on_item_double_clicked(move |id| { for item in all_items_dbl.borrow_mut().iter_mut() { item.is_editing = item.id == id; } update_view_dbl(); });

 let update_view_undo = update_view.clone(); let all_items_undo = all_items.clone(); let history_undo = history.clone();
 app.on_undo_action(move || { if let Some(prev_state) = history_undo.borrow_mut().pop() { *all_items_undo.borrow_mut() = prev_state; update_view_undo(); } });

 update_view();
 app.run()
}
