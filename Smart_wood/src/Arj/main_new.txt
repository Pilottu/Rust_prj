#![windows_subsystem = "windows"]
use slint::{ModelRc, VecModel, Timer};
use std::cell::RefCell;
use std::rc::Rc;
slint::include_modules!();
#[link(name = "user32")]
extern "system" {
    fn FindWindowW(lp_c: *const u16, lp_w: *const u16) -> *mut std::ffi::c_void;
    fn GetWindowLongW(hwnd: *mut std::ffi::c_void, n_index: i32) -> i32;
    fn SetWindowLongW(hwnd: *mut std::ffi::c_void, n_idx: i32, dw: i32) -> i32;
    fn ShowWindow(hwnd: *mut std::ffi::c_void, n_cmd_show: i32) -> i32;
    fn SetWindowPos(hwnd: *mut std::ffi::c_void, hwnd_after: *mut std::ffi::c_void, x: i32, y: i32, cx: i32, cy: i32, flags: u32) -> i32;
    fn GetAsyncKeyState(v_key: i32) -> i16;
}
const GWL_STYLE: i32 = -16;
const WS_CAPTION: i32 = 0x00C00000;
const WS_MAXIMIZEBOX: i32 = 0x00010000;
const WS_MINIMIZEBOX: i32 = 0x00020000;
const WS_THICKFRAME: i32 = 0x00040000;
const SWP_FRAMECHANGED: u32 = 0x0020;
const SWP_NOMOVE: u32 = 0x0002;
const SWP_NOSIZE: u32 = 0x0001;
const SWP_NOZORDER: u32 = 0x0004;
#[derive(Clone, Debug)]
struct RawLink {
    id: i32,
    target_node_id: i32,
    target_node_name: String,
}
#[derive(Clone, Debug)]
struct RawItem {
    id: i32,
    text: String,
    parent_id: i32,
    is_expanded: bool,
    is_editing: bool,
    note: String,
    links: Vec<RawLink>,
}
fn main() -> Result<(), slint::PlatformError> {
    let app = AppWindow::new()?;
    app.set_scale_factor(1.0_f32);
    slint::invoke_from_event_loop(move || {
        let title: Vec<u16> = "Менеджер команд голосового управления\0".encode_utf16().collect();
        unsafe {
            let hwnd = FindWindowW(std::ptr::null(), title.as_ptr());
            if !hwnd.is_null() {
                let cur = GetWindowLongW(hwnd, GWL_STYLE);
                let new_st = cur | WS_CAPTION | WS_MINIMIZEBOX | WS_MAXIMIZEBOX | WS_THICKFRAME;
                SetWindowLongW(hwnd, GWL_STYLE, new_st);
                let flags = SWP_FRAMECHANGED | SWP_NOMOVE | SWP_NOSIZE | SWP_NOZORDER;
                SetWindowPos(hwnd, std::ptr::null_mut(), 0, 0, 0, 0, flags);
                ShowWindow(hwnd, 3);
            }
        }
    }).unwrap();
    let selected_id = Rc::new(RefCell::new(1_i32));
    let all_items = Rc::new(RefCell::new(vec![RawItem {
        id: 1, text: "Основная локация".into(), parent_id: 0, is_expanded: true, is_editing: false, note: "Главный сервер управления".into(), links: vec![]
    }]));
    let history: Rc<RefCell<Vec<Vec<RawItem>>>> = Rc::new(RefCell::new(Vec::new()));
    let save_history = {
        let history = history.clone(); let all_items = all_items.clone();
        move || { let current_state = all_items.borrow().clone(); history.borrow_mut().push(current_state); }
    };
    let update_view = {
        let app_weak = app.as_weak(); let all_items = all_items.clone(); let selected_id = selected_id.clone();
        move || {
            if let Some(app) = app_weak.upgrade() {
                let db = all_items.borrow(); let mut visible_items = Vec::new(); let sel_id = *selected_id.borrow();
                
                fn build_tree(pid: i32, depth: i32, db: &Vec<RawItem>, v_items: &mut Vec<MyTreeItem>, hidden: bool, parent_lasts: Vec<bool>, sel_id: i32) {
                    let mut children: Vec<&RawItem> = db.iter().filter(|i| i.parent_id == pid).collect();
                    children.sort_by_key(|i| i.id); 
                    let total_children = children.len();
                    
                    for (index, item) in children.into_iter().enumerate() {
                        let is_last = index == total_children - 1; 
                        let has_children = db.iter().any(|i| i.parent_id == item.id);
                        
                        if !hidden {
                            let text_len = item.text.len() as f32;
                            let computed_w = (text_len * 7.2) + 20.0;
                            let last_flag = if is_last { "L" } else { "M" };
                            let smart_id = format!("{}_{}_{}_{}_{}", item.parent_id, item.id, depth, last_flag, item.links.len());
                            
                            // ПРАВИЛЬНАЯ ЛОГИКА line_mask
                            // ПРАВИЛЬНАЯ ЛОГИКА line_mask И ОПРЕДЕЛЕНИЯ КОНЦА ВЕТКИ
let mut slint_mask = [0; 6];
for i in 0..6 {
    // ВАЖНО: Линия i-го уровня должна рисоваться только если:
    // 1. Мы находимся глубже этого уровня (i < depth - 1)
    // 2. И при этом родитель на том уровне НЕ был последним (!parent_lasts[i])
    if i < (depth - 1) as usize && i < parent_lasts.len() {
        slint_mask[i] = if !parent_lasts[i] { 1 } else { 0 };
    } else {
        slint_mask[i] = 0;
    }
}

v_items.push(MyTreeItem {
    id: item.id, 
    text: item.text.clone().into(), 
    depth, 
    is_last: is_last, 
    has_children,
    // Вот эти два поля нужно вернуть:
    is_editing: item.is_editing, // или false, в зависимости от исходной логики
    is_expanded: item.is_expanded,
    parent_is_last: if depth > 1 && ((depth - 2) as usize) < parent_lasts.len() { 
    parent_lasts[(depth - 2) as usize] 
} else { 
    false 
},
    is_selected: item.id == sel_id, 
    note: item.note.clone().into(), 
    text_width: computed_w,
    smart_id: smart_id.into(), 
    line_mask: Rc::new(VecModel::from(slint_mask.to_vec())).into(),
});

                        }
                        
                        if has_children {
                            let mut current_lasts = parent_lasts.clone(); 
                            current_lasts.push(is_last);
                            build_tree(item.id, depth + 1, db, v_items, hidden || !item.is_expanded, current_lasts, sel_id);
                        }
                    }
                }
                
                build_tree(0, 0, &db, &mut visible_items, false, vec![], sel_id);
                
                let mut max_width = 180.0_f32;
                for item in &visible_items {
                    let item_width = 16.0 + (item.depth as f32 * 24.0) + 20.0 + 24.0 + 8.0 + item.text_width + 16.0; 
                    max_width = max_width.max(item_width);
                }
                app.set_tree_width(max_width.into());
                let model = Rc::new(VecModel::from(visible_items)); 
                app.set_tree_data(ModelRc::from(model));
            }
        }
    };
    update_view();
    let app_undo = app.as_weak();
    std::thread::spawn(move || loop {
        std::thread::sleep(std::time::Duration::from_millis(50));
        unsafe {
            let ctrl_down = GetAsyncKeyState(0x11) as u16 & 0x8000 != 0;
            let z_down = GetAsyncKeyState(0x5A) as u16 & 0x8000 != 0;
            if ctrl_down && z_down {
                while GetAsyncKeyState(0x5A) as u16 & 0x8000 != 0 { std::thread::sleep(std::time::Duration::from_millis(10)); }
                let app_weak_loop = app_undo.clone();
                let _ = slint::invoke_from_event_loop(move || { if let Some(app) = app_weak_loop.upgrade() { app.invoke_undo_action(); } });
            }
        }
    });
    let history_btn = history.clone(); let all_items_btn = all_items.clone(); let update_view_btn = update_view.clone();
    app.on_undo_action(move || {
        let mut db_hist = history_btn.borrow_mut();
        if let Some(previous_state) = db_hist.pop() { *all_items_btn.borrow_mut() = previous_state; update_view_btn(); }
    });
    let update_view_click = update_view.clone(); let all_items_click = all_items.clone();
    app.on_toggle_expand(move |id| {
        let mut db = all_items_click.borrow_mut(); if let Some(item) = db.iter_mut().find(|i| i.id == id) { item.is_expanded = !item.is_expanded; }
        drop(db); update_view_click();
    });
    let update_view_rename = update_view.clone(); let all_items_rename = all_items.clone();
    app.on_start_rename(move |id| {
        let mut db = all_items_rename.borrow_mut(); for item in db.iter_mut() { item.is_editing = item.id == id; }
        drop(db); update_view_rename();
    });
    let save_hist_finish = save_history.clone(); let update_view_finish = update_view.clone(); let all_items_finish = all_items.clone();
    app.on_finish_rename(move |id, new_name| {
        if new_name.is_empty() { return; } save_hist_finish();
        let mut db = all_items_finish.borrow_mut(); if let Some(item) = db.iter_mut().find(|i| i.id == id) { item.text = new_name.to_string(); item.is_editing = false; }
        drop(db); update_view_finish();
    });
    let selected_id_click = selected_id.clone(); let update_view_sel = update_view.clone();
    app.on_select_item(move |id, _text| { *selected_id_click.borrow_mut() = id; update_view_sel(); });
    let app_menu = app.as_weak();
    app.on_show_menu(move |id, x, y| {
        if let Some(app) = app_menu.upgrade() { app.set_menu_target_id(id); app.set_menu_x(x); app.set_menu_y(y); app.set_menu_visible(true); }
    });
    let all_items_note = all_items.clone(); let update_view_note = update_view.clone();
    app.on_change_note(move |id, new_note| {
        let mut db = all_items_note.borrow_mut(); if let Some(item) = db.iter_mut().find(|i| i.id == id) { item.note = new_note.to_string(); }
        drop(db); update_view_note();
    });
    let hover_timer = Rc::new(RefCell::new(Timer::default()));
    let app_hover = app.as_weak(); let all_items_hover = all_items.clone(); let timer_lock = hover_timer.clone();
    app.on_hover_item(move |id, idx, _px, _py| {
        let timer = timer_lock.clone(); timer.borrow().stop();
        if id == -1 { if let Some(ui) = app_hover.upgrade() { ui.set_tooltip_visible(false); } return; }
        let ui_weak = app_hover.clone(); let items = all_items_hover.clone(); let idx_f32 = idx as f32;
        timer.borrow_mut().start(slint::TimerMode::SingleShot, std::time::Duration::from_millis(150), move || {
            if let Some(ui) = ui_weak.upgrade() {
                let db = items.borrow();
                if let Some(item) = db.iter().find(|i| i.id == id) {
                    if !item.note.is_empty() {
                        ui.set_tooltip_text(item.note.clone().into());
                        ui.set_tooltip_index(idx_f32 as i32); ui.set_tooltip_visible(true);
                    }
                }
            }
        });
    });
    let all_items_link = all_items.clone();
    app.on_add_node_link(move |node_id, target_id| {
        let mut db = all_items_link.borrow_mut();
        if let Some(item) = db.iter_mut().find(|i| i.id == node_id) {
            let max_link_id = item.links.iter().map(|l| l.id).max().unwrap_or(0);
            item.links.push(RawLink { id: max_link_id + 1, target_node_id: target_id, target_node_name: format!("Нода {}", target_id) });
        }
    });
    let app_link_del = app.as_weak(); let all_items_del = all_items.clone();
    app.on_delete_node_link(move |node_id, link_id| {
        let mut db = all_items_del.borrow_mut();
        if let Some(item) = db.iter_mut().find(|i| i.id == node_id) {
            item.links.retain(|l| l.id != link_id);
            if let Some(ui) = app_link_del.upgrade() {
                let mut links_rc = Vec::new();
                for l in &item.links { links_rc.push(NodeLinkItem { id: l.id, target_id: l.target_node_id, target_name: l.target_node_name.clone().into() }); }
                ui.set_node_links_data(ModelRc::from(Rc::new(VecModel::from(links_rc))));
            }
        }
    });
    let save_hist_menu = save_history.clone(); let all_items_menu = all_items.clone(); let update_view_menu = update_view.clone();
    let app_menu_action = app.as_weak(); let selected_id_menu = selected_id.clone();
    app.on_menu_action(move |action_type: slint::SharedString, id: i32| match action_type.as_str() {
        "add" => {
            save_hist_menu(); let mut db = all_items_menu.borrow_mut(); let max_id = db.iter().map(|i| i.id).max().unwrap_or(0); let new_id = max_id + 1;
            if let Some(parent) = db.iter_mut().find(|i| i.id == id) { parent.is_expanded = true; }
            db.push(RawItem { id: new_id, text: "Новая локация".into(), parent_id: id, is_expanded: false, is_editing: true, note: "".into(), links: vec![] });
            *selected_id_menu.borrow_mut() = new_id; drop(db); update_view_menu(); 
        }
        "rename" => { let mut db = all_items_menu.borrow_mut(); if let Some(item) = db.iter_mut().find(|i| i.id == id) { item.is_editing = true; } drop(db); update_view_menu(); }
        "properties" => {
            let db = all_items_menu.borrow();
            if let Some(item) = db.iter().find(|i| i.id == id) {
                if let Some(ui) = app_menu_action.upgrade() {
                    ui.set_properties_text(item.note.clone().into());
                    let mut links_rc = Vec::new();
                    for l in &item.links { links_rc.push(NodeLinkItem { id: l.id, target_id: l.target_node_id, target_name: l.target_node_name.clone().into() }); }
                    ui.set_node_links_data(ModelRc::from(Rc::new(VecModel::from(links_rc))));
                    ui.set_properties_visible(true);
                }
            }
        }
        "delete" => {
            if id == 1 { return; }
            let text = { let db = all_items_menu.borrow(); db.iter().find(|i| i.id == id).map(|i| i.text.clone()).unwrap_or_default() };
            if let Some(ui) = app_menu_action.upgrade() { ui.set_confirm_text(format!("Вы уверены, что хотите удалить локацию \"{}\"?", text).into()); ui.set_confirm_visible(true); }
        }
        "confirm_delete" => {
            save_hist_menu(); let mut db = all_items_menu.borrow_mut(); 
            fn get_desc(pid: i32, db: &Vec<RawItem>, ids: &mut Vec<i32>) { 
                let children: Vec<i32> = db.iter().filter(|i| i.parent_id == pid).map(|i| i.id).collect(); 
                for cid in children { ids.push(cid); get_desc(cid, db, ids); } 
            }
            let mut to_delete = vec![id]; get_desc(id, &db, &mut to_delete); 
            db.retain(|item| !to_delete.contains(&item.id)); 
            if to_delete.contains(&*selected_id_menu.borrow()) { *selected_id_menu.borrow_mut() = 1; } 
            drop(db); update_view_menu();
        }
        _ => {}
    });
    app.run()
}
