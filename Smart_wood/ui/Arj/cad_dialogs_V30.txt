import { Button } from "std-widgets.slint";

// Два маленьких модальных диалога.

export component LinkPropertiesDialog inherits Rectangle {
    in property <int> left_id;
    in property <int> right_id;
    callback close();

    width: 100%;
    height: 100%;
    background: #000000a0;
    z: 1250;
    Rectangle {
        x: (parent.width - self.width) / 2;
        y: (parent.height - self.height) / 2;
        width: 300px;
        height: 180px;
        background: #1e1e24;
        border-radius: 8px;
        border-width: 1px;
        border-color: #ffffff20;
        VerticalLayout {
            padding: 20px;
            spacing: 10px;
            Text { text: "Свойства связи"; color: #ffd700; font-weight: 700; horizontal-alignment: center; }
            Text { text: "ID левого узла: " + root.left_id; color: white; font-size: 13px; }
            Text { text: "ID правого узла: " + root.right_id; color: white; font-size: 13px; }
            Rectangle { vertical-stretch: 1; }
            HorizontalLayout {
                alignment: end;
                Button { text: "Закрыть"; clicked => { root.close(); } }
            }
        }
    }
}

export component LinkConfirmDialog inherits Rectangle {
    callback confirm();
    callback cancel();

    width: 100%;
    height: 100%;
    background: #000000a0;
    z: 1400;
    confirm_focus := FocusScope {
        x: 0px;
        y: 0px;
        width: 0px;
        height: 0px;
        key-pressed(event) => {
            if (event.text == Key.Return) {
                root.confirm();
                return accept;
            }
            return reject;
        }
        init => { self.focus(); }
    }
    Rectangle {
        x: (parent.width - self.width) / 2;
        y: (parent.height - self.height) / 2;
        width: 320px;
        height: 150px;
        background: #1e1e24;
        border-radius: 8px;
        border-width: 1px;
        border-color: #ffaa00;
        VerticalLayout {
            padding: 20px;
            spacing: 15px;
            Text { text: "Подтвердить создание связи?"; color: #ffaa00; font-weight: 700; horizontal-alignment: center; wrap: word-wrap; }
            HorizontalLayout {
                spacing: 10px;
                alignment: center;
                Button { text: "OK (Enter)"; clicked => { root.confirm(); } }
                Button { text: "Отмена"; clicked => { root.cancel(); } }
            }
        }
    }
}
