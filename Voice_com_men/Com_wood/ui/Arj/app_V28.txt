    function handle_mid_pan(btn: PointerEventButton, kind: PointerEventKind, abs_x: length, abs_y: length) {
        if (btn == PointerEventButton.middle) {
            if (kind == PointerEventKind.down) {
                root.is_dragging_middle = true;
                root.drag_start_x = abs_x;
                root.drag_start_y = abs_y;
            }
            if (kind == PointerEventKind.up) {
                root.is_dragging_middle = false;
            }
        }
        if (kind == PointerEventKind.move && root.is_dragging_middle) {
            root.bg_offset_x += abs_x - root.drag_start_x;
            root.bg_offset_y += abs_y - root.drag_start_y;
            root.drag_start_x = abs_x;
            root.drag_start_y = abs_y;
            root.trees_geometry_changed();
        }
    }

                let old_scale = root.scale_factor;
                let new_scale = evt.delta-y > 0 ? Math.max(0.4, old_scale - 0.05) : Math.min(2.5, old_scale + 0.05);
                if (new_scale != old_scale) {
                    let cad_mouse_x = (self.mouse-x - root.bg_offset_x) / (1px * old_scale);
                    let cad_mouse_y = (self.mouse-y - root.bg_offset_y) / (1px * old_scale);

                    root.bg_offset_x = self.mouse-x - cad_mouse_x * new_scale * 1px;
                    root.bg_offset_y = self.mouse-y - cad_mouse_y * new_scale * 1px;
                }
                root.trees_geometry_changed();
            }
            accept;
        }

                root.tooltip_index = -1;

                    root.drag_start_y = self.mouse-y;

                    root.is_dragging_middle = false;
                }
            }
            if (evt.kind == PointerEventKind.move && root.is_dragging_middle) {
                root.bg_offset_x += self.mouse-x - root.drag_start_x;
                root.bg_offset_y += self.mouse-y - root.drag_start_y;

                root.drag_start_y = self.mouse-y;

                                    self.focus();
                                    self.select-all();
                                }
                            }

                                padding-left: 10px;

                                    }

                                }
                            }

                                }
                            }
                        }
                    }

                        spacing: 8px;
                        height: 32px;
                        Button {
                            text: "Удалить";
                            enabled: root.properties_notes.length > 0 && root.selected_property_note_index >= 0 && !root.note_editing;
                            clicked => {
                                root.delete_note_from_properties(root.selected_property_note_index);
                                root.selected_property_note_index = -1;
                            }
                        }
                        Rectangle { horizontal-stretch: 1; }
                        Button {
                            text: "Закрыть";
                            clicked => {
                                root.properties_visible = false;
                                root.selected_property_link_index = -1;
                                root.selected_property_note_index = -1;
                                root.note_editing = false;
                            }
                        }
                    }
                }
                if (root.active_tab == 1) : VerticalLayout {
                    spacing: 8px;
                    height: 250px;
                    if (root.properties_links.length == 0) : Text {
                        text: "Связи отсутствуют.";
                        color: #ffffff60;
                        font-size: 13px;
                    }

                            }
                            
                            VerticalLayout {
                                padding-left: 10px;
                                padding-right: 10px;
                                padding-top: 3px;
                                Text {
                                    text: "Связь ID: " + link.id;
                                    color: #ffd700;
                                    font-size: 11px;

                                    text: "→ Объект ID: " + link.target_node_id + " (" + link.target_node_name + ")";

                                    font-size: 12px;
                                }

                    HorizontalLayout {

                                root.delete_link_from_properties(root.selected_property_link_index);
                                root.selected_property_link_index = -1;

                        Rectangle { horizontal-stretch: 1; }
                        Button {
                            text: "Закрыть";
                            clicked => {
                                root.properties_visible = false;
                                root.selected_property_link_index = -1;
                                root.selected_property_note_index = -1;
                                root.note_editing = false;
                            }
                        }
                    }
                }
            }
        }
    }

    if (root.is_linking_mode && root.link_confirm_visible) : Rectangle {
        width: 100%;
        height: 100%;
        background: #000000a0;
        z: 1400;
        confirm_focus := FocusScope {
            x: 0px;
            y: 0px;
            width: 0px;
            height: 0px;
            key-pressed(event) => {
                if (event.text == Key.Return) {
                    root.finish_cad_link(root.linking_source_node_id, root.linking_source_tree_id, root.linking_target_node_id, root.linking_target_tree_id, root.linking_source_x, root.linking_source_y, root.linking_target_x, root.linking_target_y, root.linking_source_tree_x, root.linking_source_tree_y, root.linking_target_tree_x, root.linking_target_tree_y);
                    root.link_confirm_visible = false;
                    global_keys.focus();
                    return accept;
                }
                return reject;
            }
            init => { self.focus(); }
        }
        Rectangle {
            x: (parent.width - self.width) / 2;
            y: (parent.height - self.height) / 2;
            width: 320px;
            height: 150px;
            background: #1e1e24;
            border-radius: 8px;
            border-width: 1px;
            border-color: #ffaa00;
            VerticalLayout {
                padding: 20px;
                spacing: 15px;
                Text {
                    text: "Подтвердить создание связи?";
                    color: #ffaa00;
                    font-weight: 700;
                    horizontal-alignment: center;
                    wrap: word-wrap;
                }
                HorizontalLayout {
                    spacing: 10px;
                    alignment: center;
                    Button {
                        text: "OK (Enter)";
                        clicked => {
                            root.finish_cad_link(root.linking_source_node_id, root.linking_source_tree_id, root.linking_target_node_id, root.linking_target_tree_id, root.linking_source_x, root.linking_source_y, root.linking_target_x, root.linking_target_y, root.linking_source_tree_x, root.linking_source_tree_y, root.linking_target_tree_x, root.linking_target_tree_y);
                            root.link_confirm_visible = false;
                            global_keys.focus();
                        }
                    }
                    Button {
                        text: "Отмена";
                        clicked => {
                            root.link_confirm_visible = false;
                            root.linking_target_node_id = -1;
                            root.linking_target_tree_id = -1;
                            global_keys.focus();
                        }
                    }
                }
            }
        }
    }
